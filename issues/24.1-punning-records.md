# Specification
**This completely supercedes the previous.**

## Syntactic Grammar
```diff
EntryType<Named, Optional> ::= (
	| <Named+>(Word & <Optional->":") <Optional+>("?" ":") Type
+	| <Named+>("$" IDENTIFIER & <Optional+>"?")
);

PropertiesType
	::= ","? EntryType<+Named><∓Optional># ","?;

Property<Break, Return> ::= (
	| Word "=" Expression<+Block><?Break><?Return>
+	| "$" IDENTIFIER
);
```

## Decorate
```diff
Decorate(EntryType<-Named><-Optional> ::= Type) -> SemanticItemType
	:= (SemanticItemType[optional=false] Decorate(Type));
Decorate(EntryType<-Named><+Optional> ::= "?" ":" Type) -> SemanticItemType
	:= (SemanticItemType[optional=true] Decorate(Type));
Decorate(EntryType<+Named><-Optional> ::= Word ":" Type) -> SemanticPropertyType
	:= (SemanticPropertyType[optional=false]
		Decorate(Word)
		Decorate(Type)
	);
+Decorate(EntryType<+Named><-Optional> ::= "$" IDENTIFIER) -> SemanticPropertyType
+	:= (SemanticPropertyType[optional=false]
+		(SemanticKey[id=TokenWorth(IDENTIFIER)])
+		(SemanticTypeAlias[id=TokenWorth(IDENTIFIER)])
+	);
Decorate(EntryType<+Named><+Optional> ::= Word "?" ":" Type) -> SemanticPropertyType
	:= (SemanticPropertyType[optional=true]
		Decorate(Word)
		Decorate(Type)
	);
+Decorate(EntryType<+Named><+Optional> ::= "$" IDENTIFIER "?") -> SemanticPropertyType
+	:= (SemanticPropertyType[optional=true]
+		Decorate(Word)
+		Decorate(Type)
+	);

Decorate(Property ::= Word "=" Expression<+Block><?Break><?Return>) -> SemanticProperty
	:= (SemanticProperty
		Decorate(Word)
		Decorate(Expression<+Block><?Break><?Return>)
	);
+Decorate(Property ::= "$" IDENTIFIER) -> SemanticProperty
+	:= (SemanticProperty
+		(SemanticKey[id=TokenWorth(IDENTIFIER)])
+		(SemanticVariable[id=TokenWorth(IDENTIFIER)])
+	);
